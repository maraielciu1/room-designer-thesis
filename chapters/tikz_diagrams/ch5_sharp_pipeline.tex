% ch5_sharp_pipeline.tex
% fig:sharp_pipeline — SHARP inference: device → LAN POST → server → .ply → Metal viewer
% Vertical layout: client steps top, server steps middle, output bottom

\begin{tikzpicture}[
  node distance=0.5cm and 0.9cm,
  box/.style={rectangle, rounded corners=4pt, draw=black, fill=white,
              minimum width=5.0cm, minimum height=0.82cm,
              font=\small, align=center, line width=0.7pt},
  server/.style={box, fill=gray!18},
  out/.style={box, fill=black!80, text=white},
  arrow/.style={-{Stealth[length=5pt]}, thick},
  darrow/.style={arrow, dashed, draw=gray},
  lbl/.style={font=\footnotesize\itshape}
]

%%──── Client side ────
\node[box, fill=gray!8]          (cam)   {\texttt{SHARPCaptureView} — Capture JPEG};
\node[box,  below=0.5cm of cam]  (ping)  {GET /health \quad (4\,s timeout)};
\node[box,  below=0.5cm of ping] (post)  {POST /sharp/infer — JPEG bytes \quad (360\,s timeout)};

%%──── Network label ────
\node[lbl, right=0.3cm of post] {$\longrightarrow$ LAN};

%%──── Server side ────
\node[server, below=0.9cm of post]  (srv)   {Flask server \quad (\texttt{sharp\_server.py})};
\node[server, below=0.5cm of srv]   (sharp) {SHARP forward pass \quad ($\approx$10\,s)};
\node[server, below=0.5cm of sharp] (prune) {Opacity pruning \quad $\approx$1.2M $\to$ 300k Gaussians};
\node[server, below=0.5cm of prune] (resp)  {Return \texttt{.ply} \quad ($\approx$18\,MB)};

%%──── Client receives ────
\node[box, below=0.9cm of resp]  (save) {Save \texttt{sharp\_<ts>.ply} to Documents/};
\node[out, below=0.5cm of save]  (view) {Metal Splat Viewer};

%%──── Health indicator (side branch) ────
\node[box, dashed, draw=gray, fill=white, font=\footnotesize,
      right=1.2cm of ping] (ind) {Reachability\\indicator (colour)};
\draw[darrow] (ping.east) -- (ind.west);

%%──── Separation line: client vs server ────
\draw[dashed, gray, thick] (-3.2, -3.6) -- (3.2, -3.6)
  node[right, font=\footnotesize\itshape, text=gray]{device / server boundary};

%%──── Main flow arrows ────
\draw[arrow] (cam)   -- (ping);
\draw[arrow] (ping)  -- (post);
\draw[arrow] (post)  -- node[lbl, right]{LAN} (srv);
\draw[arrow] (srv)   -- (sharp);
\draw[arrow] (sharp) -- (prune);
\draw[arrow] (prune) -- (resp);
\draw[arrow] (resp)  -- node[lbl, right]{response} (save);
\draw[arrow] (save)  -- (view);

\end{tikzpicture}
